// Clock Divider Module
module clk_divider(
   input wire clk_100MHz,
   input wire reset,
   output reg clk_1Hz
);
   reg [25:0] count;

  always @(posedge clk_100MHz or posedge reset) begin
    if (reset) begin
        count <= 0;
        clk_1Hz <= 0;
    end else begin
        if (count == 26'd49_999_999) begin
            clk_1Hz <= ~clk_1Hz;
            count <= 0;
        end else begin
            count <= count + 1;
        end
    end
  end
endmodule

   input wire clk_1Hz,
   input wire reset,
   input wire up_down,        // 1 = up, 0 = down

   always @(posedge clk_1Hz or posedge reset) begin
      if (reset)
          count <= 4'b0000;
      else if (up_down)

reg [25:0] count;          // 26-bit counter for division

always @(posedge clk_100MHz) begin
  if (rst) begin
      count <= 0;
      clk_1hz <= 0;
  end
  else begin
      if (count == 26'd49000000) begin
          clk_1hz <= ~clk_1hz; // Toggle output clock
          count <= 0;
      end
      else begin
          count <= count + 1;

  end
