//Top Module
module adder_display(
   input [3:0] A, // 4-bit input A
   input [3:0] B, // 4-bit input B
   input     Cin, // Carry-in
   output [6:0] display, // 7-segment display output
   output [7:0] an, // Anode control (only 1 digit active)
   output     dp // Decimal point
);

  wire [4:0] sum; // 5-bit result to hold sum and carry-out

  // Perform addition
  assign sum = A + B + Cin;

  // Instantiate seven-segment display encoder
  sevseg1 ssd (
      .digit(sum[3:0]), // Only lower 4 bits sent to display
      .display(display),
      .an(an),
      .dp(dp)
  );

endmodule

   output wire [7:0] an,
   output wire dp

  // Segment values for common anode
  parameter zero = 7'b0000001;
