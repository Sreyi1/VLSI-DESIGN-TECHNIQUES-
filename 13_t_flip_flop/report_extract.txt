module tff (
   input wire t,
   input wire clk,
   input wire reset,
   output reg q
);
   always @(posedge clk or posedge reset) begin
      if (reset)
          q <= 0;
      else if (t)
          q <= ~q; // Toggle output if t = 1
      // else retain current q
   end
endmodule
