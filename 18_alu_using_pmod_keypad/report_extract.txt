module clk_dvd(
   input wire clk_100MHz, // 100 MHz input clock
   input wire rst,     // Active high reset
   output reg clk_1hz      // Output 1 Hz clock
);

reg [25:0] count;          // 26-bit counter for division

always @(posedge clk_100MHz) begin
  if (rst) begin
      count <= 0;
      clk_1hz <= 0;
  end
  else begin
      if (count == 26'd49000000) begin
          clk_1hz <= ~clk_1hz; // Toggle output clock
          count <= 0;
      end
      else begin
          count <= count + 1;
      end
  end
end

endmodule

module clk_divider (
   input clkin,
   input rset,
   output clkout
);

  reg [1:0] cur_state, next_state;
  parameter s1 = 2'b00;
  parameter s2 = 2'b01;
  parameter s3 = 2'b10;

  // State register
  always @(posedge clkin or posedge rset) begin
     if (rset)
         cur_state <= s2;
     else
         cur_state <= next_state;
  end

  // Next state logic
  always @(*) begin
     case (cur_state)
       s1: next_state = s2;
       s2: next_state = s3;
       s3: next_state = s1;
       default: next_state = s1;
     endcase
  end

  // Output logic
  assign clkout = (cur_state == s3) ? 1'b1 : 1'b0;
