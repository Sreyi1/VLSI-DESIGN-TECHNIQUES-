module jkff (
   input wire j,
   input wire k,
   input wire clk,
   input wire reset,
   output reg q
);
   always @(posedge clk or posedge reset) begin
      if (reset)
          q <= 0;
      else begin
          case ({j, k})
            2'b00: q <= q;  // No change
            2'b01: q <= 0;  // Reset
            2'b10: q <= 1;  // Set
            2'b11: q <= ~q;  // Toggle
          endcase
      end
   end
endmodule

module jkff_tb;

  reg j, k;
  reg clk, reset;
  wire q;

  jkff uut (
     .j(j),
     .k(k),
     .clk(clk),
     .reset(reset),
     .q(q)
  );

  initial begin
     clk = 0;
     forever #5 clk = ~clk; // 10ns clock period
  end

  initial begin
     // Initialize
     j = 0; k = 0; reset = 1;
     #10 reset = 0;

     // Test cases
     #10 j = 0; k = 0; // No change
     #10 j = 1; k = 0; // Set
     #10 j = 0; k = 1; // Reset
     #10 j = 1; k = 1; // Toggle
     #10 j = 1; k = 1; // Toggle again

    #10 $stop;
  end
endmodule
